  <action>
    1. Create nextjs/lib/auth-guard.ts with the following:

    ```typescript
    import { NextResponse } from 'next/server';
    import { cookies } from 'next/headers';
    import { db as prisma } from '@/lib/db';

    interface VerifyResult {
      authorized: boolean;
      response?: NextResponse;
    }

    /**
     * Re-verify user's admin role from DB for critical write operations.
     * If the user was demoted since their JWT was issued, force re-authentication
     * by deleting session cookies and returning 401 (per D-04).
     */
    export async function verifyAdminRole(
      userEmail: string,
      projectId: string
    ): Promise<VerifyResult> {
      // Check superadmin status from DB (not JWT)
      const dbUser = await prisma.user.findUnique({
        where: { email: userEmail },
        select: { isSuperAdmin: true },
      });
      if (dbUser?.isSuperAdmin) {
        return { authorized: true };
      }

      // Check project membership role from DB
      const membership = await prisma.projectMember.findUnique({
        where: {
          projectId_userEmail: {
            projectId,
            userEmail,
          },
        },
        select: { role: true },
      });

      if (!membership || (membership.role !== 'admin' && membership.role !== 'owner')) {
        // User was demoted or removed - force re-authentication
        const cookieStore = await cookies();
        cookieStore.delete('authjs.session-token');
        cookieStore.delete('__Secure-authjs.session-token');

        return {
          authorized: false,
          response: NextResponse.json(
            { error: 'Session expired - role changed', code: 'ROLE_CHANGED' },
            { status: 401 }
          ),
        };
      }

      return { authorized: true };
    }
    ```

    Important: Delete BOTH cookie variants (authjs.session-token and __Secure-authjs.session-token) per Pitfall 2 in research.
